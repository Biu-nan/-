39:   const p = (n) => String(n).padStart(2, '0');

65: /**
66:  * 幂等写入事实。返回 { written, skipped, unchanged, total }
67:  * facts[]: { source, storeId, caliber, subject, window, statDate, value, unit, reason?, meta? }
68:  */
69: export async function appendFacts(facts, { date } = {}) {
70:   const schema = await loadSchema();
71:   const byDate = new Map();
72: 
73:   for (const f of facts) {
74:     assertKnownCaliber(schema, f.caliber);
75:     if (!f.statDate) f.statDate = date || localDate();
76:     if (!f.source) throw new Error('fact 缺少 source');
77:     const d = f.statDate;
78:     if (!byDate.has(d)) byDate.set(d, []);
79:     byDate.get(d).push(f);
80:   }

148: /** 按口径取值（找不到返回 null，绝不猜） */
149: export function pick(facts, { source, caliber, subject = '-', window = '-', storeId }) {
150:   for (const f of facts) {
151:     if (f.source !== source) continue;
152:     if (f.caliber !== caliber) continue;
153:     if ((f.subject || '-') !== subject) continue;
154:     if ((f.window || '-') !== window) continue;
155:     if (storeId && f.storeId !== storeId) continue;
156:     return f;
157:   }
